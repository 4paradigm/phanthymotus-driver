cmake_minimum_required(VERSION 3.16)
project(franka_state_reader LANGUAGES CXX)
find_package(Franka 0.18 REQUIRED)
find_package(Threads REQUIRED)
add_executable(franka-state-reader read_state.cpp)
target_compile_features(franka-state-reader PRIVATE cxx_std_17)
target_compile_options(franka-state-reader PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(franka-state-reader PRIVATE Franka::Franka Threads::Threads)
install(TARGETS franka-state-reader RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(franka-joint-motion-test test_joint_motion.cpp)
  target_compile_features(franka-joint-motion-test PRIVATE cxx_std_17)
  add_test(NAME franka-joint-motion COMMAND franka-joint-motion-test)
endif()
