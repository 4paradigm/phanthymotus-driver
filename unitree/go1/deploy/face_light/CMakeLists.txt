cmake_minimum_required(VERSION 3.10)
project(go1_face_light_adapter LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The official faceLight SDK libraries require Linux")
endif()
set(FACE_LIGHT_SDK_DIR "" CACHE PATH "Verified vendor faceLightSDK_Nano directory")
if(NOT EXISTS "${FACE_LIGHT_SDK_DIR}/include/FaceLightClient.h")
  message(FATAL_ERROR "Provide FACE_LIGHT_SDK_DIR with the verified vendor SDK")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(SDK_ARCH arm64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  set(SDK_ARCH amd64)
else()
  message(FATAL_ERROR "Unsupported SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(SDK_LIBRARY "${FACE_LIGHT_SDK_DIR}/lib/libfaceLight_SDK_${SDK_ARCH}.so")
if(NOT EXISTS "${SDK_LIBRARY}")
  message(FATAL_ERROR "Missing official SDK library: ${SDK_LIBRARY}")
endif()
add_executable(face_light_sdk_adapter adapter.cpp)
target_compile_features(face_light_sdk_adapter PRIVATE cxx_std_11)
target_compile_options(face_light_sdk_adapter PRIVATE -Wall -Wextra -Werror)
target_include_directories(face_light_sdk_adapter PRIVATE "${FACE_LIGHT_SDK_DIR}/include")
add_library(face_light_vendor SHARED IMPORTED)
set_target_properties(face_light_vendor PROPERTIES IMPORTED_LOCATION "${SDK_LIBRARY}" IMPORTED_NO_SONAME TRUE)
target_link_libraries(face_light_sdk_adapter PRIVATE face_light_vendor ${CMAKE_DL_LIBS})
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/vendor_sdk")
configure_file("${SDK_LIBRARY}" "${CMAKE_BINARY_DIR}/vendor_sdk/libfaceLight_SDK_${SDK_ARCH}.so" COPYONLY)
# Export sendto for vendor SDK PLT resolution; fail closed if it is not observed.
set_target_properties(face_light_sdk_adapter PROPERTIES LINK_FLAGS "-Wl,--export-dynamic"
  BUILD_RPATH "$ORIGIN/vendor_sdk")
